-- agregar-horarios-especiales.sql -- -- Pega esto completo en el SQL Editor de Supabase (una query nueva). -- -- Crea la tabla horarios_especiales: aperturas puntuales para una fecha -- concreta que el patrón semanal de horarios_laborales no contempla (ej. -- un sábado especial en un negocio que normalmente no abre sábados). Se -- SUMAN al patrón semanal para ese día, no lo reemplazan. Un solo bloque -- por fecha por recurso (no admite horario partido dentro de una apertura -- especial, a diferencia de horarios_laborales). -- -- Como es una tabla nueva, este archivo también incluye sus políticas RLS -- completas (lectura para el panel + escritura de insertar/borrar) — no -- hace falta tocar rls-panel-auth.sql ni ningún otro archivo ya ejecutado. create table horarios_especiales ( id uuid primary key default gen_random_uuid(), recurso_id uuid not null references recursos(id) on delete cascade, fecha date not null, hora_inicio time not null, hora_fin time not null, check (hora_fin > hora_inicio), unique (recurso_id, fecha) ); alter table horarios_especiales enable row level security; create policy "panel_leer_horarios_especiales" on horarios_especiales for select using (auth.role() = 'authenticated'); create policy "panel_escribir_horarios_especiales" on horarios_especiales for insert with check (auth.role() = 'authenticated'); create policy "panel_borrar_horarios_especiales" on horarios_especiales for delete using (auth.role() = 'authenticated');
